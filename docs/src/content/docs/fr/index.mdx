import { Card, CardGrid } from '@astrojs/starlight/components';

:::caution[Pré-alpha]
Seul le profil **observe** existe aujourd'hui. Syngard détecte et signale la dérive ; il n'empêche pas encore un agent de lire des instructions non approuvées.
:::

<CardGrid stagger>
	<Card title="Au bit près" icon="seti:lock">
		SHA-256 sur les octets bruts, sans normalisation. Un octet modifié, un script ajouté ou un fichier supprimé change le digest du bundle.
	</Card>
	<Card title="Résultats honnêtes" icon="warning">
		Une inspection incomplète (symlink, fichier spécial, limite de taille, entrée illisible) est signalée `unknown`, jamais comme conforme.
	</Card>
	<Card title="Inerte par conception" icon="approve-check">
		Les fichiers inspectés sont des données hostiles : jamais exécutés, jamais téléchargés, jamais suivis. Les noms hostiles sont échappés à l'affichage.
	</Card>
	<Card title="Scriptable" icon="seti:json">
		Codes de sortie stables et rapports JSON versionnés avec schémas publiés, prêts pour la CI.
	</Card>
</CardGrid>
